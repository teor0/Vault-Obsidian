Cambio coordinate: $$\begin{align}
dV=rdrd\phi dz \\
dV=R^2sen\theta dRd\phi d\theta
\end{align}$$
Prodotto vettoriale versori: $$\begin{align}
\hat{x}\times\hat{x}=\hat{y}\times\hat{y}=\hat{y}\times\hat{y}=0 \\
\hat{x}\times\hat{y}=\hat z \\
\hat{y}\times\hat{z}=\hat x \\
\hat{x}\times\hat{z}=\hat y \\
\hat{y}\times\hat{x}=\hat{-z} \\
\hat{z}\times\hat{y}=\hat{-x} \\
\hat{z}\times\hat{x}=\hat{-y} \\
\end{align}$$
Prodotto vettoriale: $$A\times B=A*B*sen(\theta)=\begin{bmatrix}i\ j\ k\\ A_x A_y A_z \\ B_x B_y B_z \end{bmatrix}=\begin{align}\hat{i}[Ay*B_z-(A_z*B_y)]\\ -\hat{j}[Ax*B_z-(A_z*B_x)]+\\ \hat{k}[Ax*B_y-(A_y*B_x)] \end{align}$$
Campo elettrico: $$\begin{align}
\underline{E}={\hat{i}_R \cdot {q_1\over 4\pi \varepsilon R^2}} \\
\underline{E}_{TOT}={1\over 4\pi\varepsilon}\cdot \sum^{n}_{i=1}{q_i(\underline{R}-\underline{R_i})\over |\underline{R}-\underline{R_i}|^3} \\
\hat{i}_R={\underline{R}-\underline{R_1}\over |\underline{R}-\underline{R_1}|} \\ 
\varepsilon=\varepsilon_r\cdot\varepsilon_0 \\
\underline{E}=-\nabla V \\
\underline{E}={\underline{J}\over \sigma} \\
\underline{E}={\rho_l\over 2\pi\varepsilon_0 r}
\end{align}$$
Forza elettrica: $$F_E={\hat{i}_R \cdot {q_1q_2\over 4\pi \varepsilon R^2}}=q\cdot \underline{E}$$
Induzione elettrica: $$\underline{D}=\varepsilon\underline{E}$$
Legge di Gauss: $$\begin{align}
Q={\huge\unicode{x222F}}_{S}\ \underline{D}\ \circ\ \hat{u}\ dS \\
D={Q\over A}\ \text{in modulo}
\end{align}$$
Capacità totale: $$\begin{align}
Q=\int_S \rho_S dS\\
Q=C\cdot V 
\end{align}$$
Potenziale: $$\begin{align}
V(\underline{R})=\sum^{n}_{i=1}{q\over 4\pi\varepsilon |R-R_i|}\\
V={q\over 4\pi\varepsilon R}
\end{align}$$
Variazione di energia potenziale: $$\begin{align}
\omega=-\int q\underline{E}\circ d\underline{l}=q\cdot V
\end{align}$$
Differenza di potenziale o tensione: $$\begin{align}
V_{21}=V_2-V_1=\int_l -\underline{E}\circ d\underline{l} \\
V=RI
\end{align}$$
Densità di corrente: $$\begin{align}
\underline{J}=\rho_S\cdot\underline{V}_e\\
\underline{J}=\sigma\cdot\underline{E}
\end{align}$$
Corrente:$$\begin{align}
I=\int_S \underline{J}\circ\hat{u}\ dS
\end{align}$$
Resistenza:$$\begin{align}
R={V\over I}\\
R={l\over \sigma\cdot A}
\end{align}$$
Potenza dissipata: $$\begin{align}
P=\iiint_V \underline{E}\cdot\underline{J} dV \\
P=\iiint_V \sigma |\underline{E}|^2dV \\
P=I^2R \\
P=IV
\end{align}$$
Energia potenziale elettrostatica: $$\begin{align}
\omega={1\over 2}\large\int \varepsilon\cdot\underline{E}^2\cdot dV \\
\omega_e={1\over 2}CV^2
\end{align}$$
Forza magnetica: $$\begin{align}
\underline{F}_m=I(\underline{l}\times\underline{B})
\end{align}$$
Campo magnetico/ legge di Biot-Savart: $$\begin{align}
\underline{H}={I\over 4\pi}\int_l{d\underline{l}\times\hat{i}_R\over R^2} dz \\
\underline{H}=\hat{\phi}{I\over 2\pi r}
\end{align}$$
Legge di Ampere: $$\begin{align}
I=\oint_C\underline{H}d\underline{l}
\end{align}$$
Capacità di un condensatore:$$\begin{align}
C={Q\over V}={\varepsilon\cdot A\over d} \\
\text{parallelo}\ C=C_1+C_{2} \\
\text{serie}\ C={C_1C_2\over C_1+C_2}
\end{align}$$

f_{em}=-{\sigma\over\sigma t}\phi_{b} \\
